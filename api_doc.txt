
#USER calls

http://localhost/api/users
type: GET
Json: no json

http://localhost/api/users/{fullName}
type: GET
Json: no json

http://localhost/api/users/add
type: POST
Json: 
{
    "fullName": 
    "address":
    "email": 
    "username": 
    "password":
    "role": "CLIENT"/"ADMIN"
}

http://localhost/api/users/{id}/update
type: PUT
Json: 
{
    "fullName":, 
    "address":,
    "email":,
    "username":,
    "password":,
    "newPassword": (could be missing or empty),
    "role": "CLIENT"/"ADMIN"
}

http://localhost/api/users/{id}/delete
type: DELETE
Json: no json



#THE RESPONSE JSON HAS:

 UUID id;

 String fullName;

 String address;

 LocalDateTime createdDate;

 LocalDateTime updatedDate;

 String email;

#CREDENTIALS calls

http://localhost/api/credentials
type: GET
Json: no json

http://localhost/api/credentials/{id} --> {id} is the user id
type: GET
Json: no json

http://localhost/api/credentials/add
type: POST
Json: 
{
  username: ,
  password: ,
  role:
}

http://localhost/api/credentials/{id}/update
type: PUT
Json: 
{
  username: ,
  password: ,
  role:
}

http://localhost/api/credentials/{id}/delete
type: DELETE
Json: no json



#THE RESPONSE JSON HAS:
For general requests:
 UUID id;

 String username;

 String role;

AND

For login request:
 String token
 UUID id 
 String username
 String role
